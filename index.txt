  function calcularTransfer(){
    const km = parseFloat(document.getElementById('ct_km').value) || 0;
    const regressoVazio = document.getElementById('regressoVazio').checked;
    const isVan = state.veiculoT === 'van';

    document.getElementById('ct_energia_display').textContent = `€${energia.toFixed(2)}`;

    const curtoMin = isVan ? cfg.ct_curto_min_van : cfg.ct_curto_min_sedan;
    const kmRate = isVan ? cfg.ct_km_rate_van : cfg.ct_km_rate_sedan;
    const horaRate = isVan ? cfg.ct_hora_van : cfg.ct_hora_sedan;
    const taxaEspera = isVan ? cfg.espera_van : cfg.espera_sedan;

    const valorKm = km <= 20 ? curtoMin : km * kmRate;

    let esperaMin = 0, valorServico;
    if(regressoVazio){
      const duracaoIdaMin = parseFloat(document.getElementById('ct_duracaoIda').value) || 0;
      const duracaoVoltaMin = parseFloat(document.getElementById('ct_duracaoVolta').value) || 0;
      esperaMin = parseFloat(document.getElementById('ct_esperaDupla').value) || 0;
      const horaExtra = (duracaoIdaMin/60)*horaRate + (duracaoVoltaMin/60)*horaRate*(cfg.ct_volta_pct/100);
      valorServico = Math.max(valorKm, horaExtra);
    } else {
      const tempoMin = parseFloat(document.getElementById('ct_tempo').value) || 0;
      esperaMin = parseFloat(document.getElementById('ct_espera').value) || 0;
      const valorTempo = (tempoMin/60) * horaRate;
      valorServico = Math.max(valorKm, valorTempo);
    }

    const esperaValor = (esperaMin/60) * taxaEspera;
    const valorMinimo = valorServico;
    let valorRecomendado = valorMinimo * (1 + cfg.ct_margem_pct/100) + portagem + energia + esperaValor + bagagemVal + cadeiraVal;
    if(comRecibo) valorRecomendado *= (1 + cfg.iva_pct/100);

    document.getElementById('ct_valorRecomendado').textContent = `€${valorRecomendado.toFixed(2)}`;

  function pad(n){ return String(n).padStart(2,'0'); }

  function escapeICS(txt){
    return String(txt).replace(/\\/g,'\\\\').replace(/;/g,'\\;').replace(/,/g,'\\,').replace(/\n/g,'\\n');
  }

  function gerarICS(ag){
    const [ano,mes,dia] = ag.data.split('-').map(Number);
    const [hora,min] = ag.hora.split(':').map(Number);
    const inicio = new Date(ano, mes-1, dia, hora, min);
    const fim = new Date(inicio.getTime() + 60*60*1000);
    const fmt = (d)=> `${d.getFullYear()}${pad(d.getMonth()+1)}${pad(d.getDate())}T${pad(d.getHours())}${pad(d.getMinutes())}00`;
    const agora = new Date();
    const dtstamp = `${agora.getUTCFullYear()}${pad(agora.getUTCMonth()+1)}${pad(agora.getUTCDate())}T${pad(agora.getUTCHours())}${pad(agora.getUTCMinutes())}${pad(agora.getUTCSeconds())}Z`;
    const desc = `Partida: ${ag.partida}\nChegada: ${ag.chegada}\nValor: €${ag.valor}${ag.fornecedor ? '\nFornecedor: ' + ag.fornecedor : ''}`;

    return [
      'BEGIN:VCALENDAR',
      'VERSION:2.0',
      'PRODID:-//Trajeto//Agendamento//PT',
      'BEGIN:VEVENT',
      `UID:${ag.id}@trajeto.app`,
      `DTSTAMP:${dtstamp}`,
      `DTSTART:${fmt(inicio)}`,
      `DTEND:${fmt(fim)}`,
      `SUMMARY:${escapeICS('Transfer: ' + ag.cliente)}`,
      `DESCRIPTION:${escapeICS(desc)}`,
      `LOCATION:${escapeICS(ag.partida)}`,
      'BEGIN:VALARM',
      'ACTION:DISPLAY',
      `DESCRIPTION:${escapeICS('Lembrete: transfer de ' + ag.cliente + ' amanhã')}`,
      'TRIGGER:-P1D',
      'END:VALARM',
      'END:VEVENT',
      'END:VCALENDAR'
    ].join('\r\n');
  }

  function abrirICS(ag){
    const conteudo = gerarICS(ag);
    const blob = new Blob([conteudo], {type:'text/calendar;charset=utf-8'});
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `trajeto-${ag.cliente.replace(/\s+/g,'_')}-${ag.data}.ics`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    setTimeout(()=> URL.revokeObjectURL(url), 4000);
  }

  function normalizarTelefone(tel){
    let d = String(tel).replace(/[^\d+]/g,'');
    if(d.startsWith('+')) d = d.slice(1);
    if(d.startsWith('00')) d = d.slice(2);
    if(d.length === 9) d = '351' + d; // nº português sem indicativo
    return d;
  }

  function inicioSemana(d){
    const dia = new Date(d);
    const diaSemana = (dia.getDay() + 6) % 7; // 0 = segunda
    dia.setDate(dia.getDate() - diaSemana);
    dia.setHours(0,0,0,0);
    return dia;
  }

  function renderFaturamento(){
    const lista = lerAgendamentos().filter(a => a.realizado);
    const hoje = new Date();
    const hojeStr = hoje.toISOString().slice(0,10);
    const inicioSem = inicioSemana(hoje);
    const mesAtual = hojeStr.slice(0,7);

    let totalHoje = 0, totalSemana = 0, totalMes = 0;
    lista.forEach(ag=>{
      const v = parseFloat(ag.valor) || 0;
      const dataAg = new Date(ag.data + 'T00:00:00');
      if(ag.data === hojeStr) totalHoje += v;
      if(dataAg >= inicioSem) totalSemana += v;
      if(ag.data.slice(0,7) === mesAtual) totalMes += v;
    });

    document.getElementById('fat_hoje').textContent = `€${totalHoje.toFixed(0)}`;
    document.getElementById('fat_semana').textContent = `€${totalSemana.toFixed(0)}`;
    document.getElementById('fat_mes').textContent = `€${totalMes.toFixed(0)}`;

    const el = document.getElementById('listaFaturamento');
    const ordenada = [...lista].sort((a,b)=> (b.data+b.hora).localeCompare(a.data+a.hora));
    if(ordenada.length === 0){
      el.innerHTML = '<p class="ag-vazio">Ainda não tens corridas faturadas. Marca um agendamento como "✅ Concluído".</p>';
      return;
    }
    el.innerHTML = ordenada.map(ag => `
      <div class="ag-item">
        <div class="ag-trajeto">${ag.partida} → ${ag.chegada}</div>
        <div class="ag-meta">
          📅 ${ag.data.split('-').reverse().join('/')}<br>
          👤 ${ag.cliente} &nbsp;·&nbsp; <span class="ag-valor">€${ag.valor}</span>
          ${ag.fornecedor ? `<br>🏷 Fornecedor: ${ag.fornecedor}` : ''}
        </div>
      </div>
    `).join('');
  }

  document.getElementById('btnGuardarAgendamento').addEventListener('click', ()=>{
    const partida = document.getElementById('ag_partida').value.trim();

      return;
    }

    const ag = {
      id: Date.now().toString(36) + Math.random().toString(36).slice(2,7),
      partida, chegada, data, hora, cliente, telefone, fornecedor,
      valor: valor || document.getElementById('ct_cobrar').value,
      realizado: false
    };

    const lista = lerAgendamentos();
    lista.push(ag);
    gravarAgendamentos(lista);
    renderLista();

    document.getElementById('ag_partida').value = '';
    document.getElementById('ag_chegada').value = '';
    document.getElementById('ag_cliente').value = '';
    document.getElementById('ag_telefone').value = '';
